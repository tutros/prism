```python
"""Server-Sent Events endpoint for progress streaming."""

from fastapi import APIRouter
from sse_starlette.sse import EventSourceResponse
import asyncio
import json
from typing import AsyncGenerator

router = APIRouter(prefix="/api", tags=["progress"])

# In-memory progress store (comparison_id -> list of status messages)
# For POC; production would use Redis or similar
progress_store: dict[str, list[str]] = {}
progress_complete: dict[str, bool] = {}


def add_progress(comparison_id: str, status: str) -> None:
    """Add a progress message for a comparison."""
    if comparison_id not in progress_store:
        progress_store[comparison_id] = []
    progress_store[comparison_id].append(status)


def mark_complete(comparison_id: str) -> None:
    """Mark a comparison as complete."""
    progress_complete[comparison_id] = True


async def progress_generator(comparison_id: str) -> AsyncGenerator[dict, None]:
    """Generate SSE events for a comparison's progress."""
    last_index = 0
    max_wait = 120  # 2 minute timeout
    waited = 0

    while waited < max_wait:
        # Check for new messages
        messages = progress_store.get(comparison_id, [])
        while last_index < len(messages):
            yield {
                "event": "progress",
                "data": json.dumps({"status": messages[last_index]})
            }
            last_index += 1

        # Check if complete
        if progress_complete.get(comparison_id):
            yield {
                "event": "complete",
                "data": json.dumps({"comparison_id": comparison_id})
            }
            return

        # Wait and poll again
        await asyncio.sleep(0.5)
        waited += 0.5

    # Timeout
    yield {
        "event": "error",
        "data": json.dumps({"error": "Progress stream timed out"})
    }


@router.get("/progress/{comparison_id}")
async def stream_progress(comparison_id: str):
    """Stream progress updates for a comparison via SSE."""
    return EventSourceResponse(progress_generator(comparison_id))
```

```typescript
import { useEffect, useState, useCallback } from 'react'

interface ProgressState {
  status: string
  error: string | null
  complete: boolean
  connected: boolean
}

export function useProgress(comparisonId: string | null) {
  const [state, setState] = useState<ProgressState>({
    status: 'Initializing...',
    error: null,
    complete: false,
    connected: false,
  })

  const resetProgress = useCallback(() => {
    setState({
      status: 'Initializing...',
      error: null,
      complete: false,
      connected: false,
    })
  }, [])

  useEffect(() => {
    if (!comparisonId) return

    // Reset state for new comparison
    resetProgress()

    const eventSource = new EventSource(`/api/progress/${comparisonId}`)

    eventSource.onopen = () => {
      setState(prev => ({ ...prev, connected: true }))
    }

    eventSource.addEventListener('progress', (event) => {
      try {
        const data = JSON.parse(event.data)
        setState(prev => ({ ...prev, status: data.status }))
      } catch (e) {
        console.error('Failed to parse progress event:', e)
      }
    })

    eventSource.addEventListener('complete', () => {
      setState(prev => ({ ...prev, complete: true, status: 'Complete' }))
      eventSource.close()
    })

    eventSource.addEventListener('error', (event) => {
      // Check if it's a custom error event with data
      if (event instanceof MessageEvent && event.data) {
        try {
          const data = JSON.parse(event.data)
          setState(prev => ({ ...prev, error: data.error }))
        } catch (e) {
          setState(prev => ({ ...prev, error: 'Connection error' }))
        }
      }
      eventSource.close()
    })

    eventSource.onerror = () => {
      // Connection error - EventSource will auto-reconnect by default
      // After 3 failed reconnects, give up
      setState(prev => {
        if (!prev.connected) {
          // Never connected - fall back to polling
          return { ...prev, error: 'Unable to connect to progress stream' }
        }
        return prev
      })
    }

    return () => {
      eventSource.close()
    }
  }, [comparisonId, resetProgress])

  return state
}
```

```typescript
interface ProgressDisplayProps {

export function ProgressDisplay({ status, connected }: ProgressDisplayProps) {
  return (
    <div className="flex flex-col items-center justify-center py-12 space-y-4">
      {/* Spinner */}
      <div className="animate-spin rounded-full h-10 w-10 border-b-2 border-blue-600" />

      {/* Status text */}
      <div className="text-center">
        <p className="text-gray-700 font-medium">{status}</p>
        {!connected && (
          <p className="text-xs text-gray-500 mt-1">Connecting...</p>
        )}
      </div>

      {/* Progress stages indicator */}
      <div className="flex items-center space-x-2 text-xs text-gray-400">
        <StageIndicator active={status.includes('Searching')} label="Search" />
        <span>-</span>
        <StageIndicator active={status.includes('Analyzing') || status.includes('Found')} label="Analyze" />
        <span>-</span>
        <StageIndicator active={status.includes('Generated') || status.includes('Finalizing')} label="Generate" />
      </div>
    </div>
  )
}

function StageIndicator({ active, label }: { active: boolean; label: string }) {
  return (
    <span className={active ? 'text-blue-600 font-medium' : ''}>
      {label}
    </span>
  )
}
```
