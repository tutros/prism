"""Claude API service for criteria generation."""

from pydantic import BaseModel, Field

from .config import get_anthropic_client
from .tavily_service import OptionResearch


class CriterionOutput(BaseModel):
    """Single criterion generated by Claude."""
    name: str = Field(description="Short name for the criterion")
    description: str = Field(description="Explanation of why this criterion matters")
    score_a: float = Field(ge=0, le=100, description="Score for option A (0-100)")
    score_b: float = Field(ge=0, le=100, description="Score for option B (0-100)")
    weight: float = Field(ge=0, le=100, description="Importance weight based on user intent (0-100)")
    sources: list[str] = Field(description="Source URLs from research that support this criterion")


class CriteriaOutput(BaseModel):
    """Full criteria generation output."""
    criteria: list[CriterionOutput] = Field(min_length=3, max_length=5)
    winner: str = Field(description="'a', 'b', or 'tie' based on weighted scores")
    reasoning: str = Field(description="2-3 sentence explanation of why the winner won")


async def generate_criteria(
    option_a: str,
    option_b: str,
    intent: str,
    research_a: OptionResearch,
    research_b: OptionResearch,
) -> CriteriaOutput:
    """Generate non-obvious criteria from research using Claude structured outputs.

    Uses Claude's structured outputs feature for guaranteed schema compliance.
    Implements prompt caching for the system prompt to reduce costs by 90%.
    """
    client = get_anthropic_client()

    # Format research into readable context
    context_a = "\n\n".join([
        f"**{r.title}** ({r.url})\n{r.content}"
        for r in research_a.results
    ]) if research_a.results else "No research results found."

    context_b = "\n\n".join([
        f"**{r.title}** ({r.url})\n{r.content}"
        for r in research_b.results
    ]) if research_b.results else "No research results found."

    # Collect all source URLs for citation validation
    all_sources = {r.url for r in research_a.results} | {r.url for r in research_b.results}

    prompt = f"""Analyze the research below and generate 3-5 NON-OBVIOUS criteria for comparing "{option_a}" vs "{option_b}".

USER INTENT: {intent}

CRITICAL RULES:
1. Focus on HIDDEN COSTS - things people don't normally consider:
   - Embedded energy and carbon footprint of production
   - Government subsidies that distort true costs
   - Externalities (pollution, health impacts, social costs)
   - Supply chain transparency and labor practices
   - Long-term systemic impacts and second-order effects

2. DO NOT include obvious factors like:
   - Direct purchase price
   - Basic durability or quality
   - Convenience or ease of use
   - Brand reputation

3. Each criterion MUST include at least one source URL from the research below.
   Only use URLs that appear in the research - do not make up URLs.

4. Weight criteria based on the user's stated intent: "{intent}"
   Higher weights for criteria more relevant to their priorities.

5. Score both options 0-100 for each criterion based on the research.
   Higher score = better performance on that criterion.

RESEARCH FOR "{option_a}":
{context_a}

RESEARCH FOR "{option_b}":
{context_b}

AVAILABLE SOURCE URLS (only use these):
{chr(10).join(sorted(all_sources))}

Generate exactly 3-5 criteria with scores, weights, and source citations."""

    response = await client.messages.create(
        model="claude-sonnet-4-5-20250929",
        max_tokens=4096,
        system=[
            {
                "type": "text",
                "text": "You are an expert at discovering non-obvious decision criteria that surface hidden costs and systemic impacts.",
            },
            {
                "type": "text",
                "text": """Your role is to help people see the invisible threads connecting everyday choices to larger systems.

Examples of excellent non-obvious criteria:
- "Embedded Water Usage" - water consumed in production, not just in use
- "Subsidy Distortion" - how government subsidies mask true costs
- "End-of-Life Complexity" - difficulty and environmental cost of disposal/recycling
- "Supply Chain Labor Standards" - working conditions throughout production
- "Carbon Lock-in Effect" - how choosing this option affects future emissions

Always ground your analysis in the provided research. Never hallucinate facts or sources.""",
                "cache_control": {"type": "ephemeral"}  # Cache this for 90% cost reduction
            }
        ],
        messages=[{"role": "user", "content": prompt}],
    )

    # Parse response - Claude returns JSON when we specify structured output via tools
    # For now, use tool_use pattern for structured output
    import json

    # Extract text content and parse as JSON
    content = response.content[0].text

    # Find JSON in the response (Claude often wraps it in markdown code blocks)
    if "```json" in content:
        json_str = content.split("```json")[1].split("```")[0].strip()
    elif "```" in content:
        json_str = content.split("```")[1].split("```")[0].strip()
    else:
        json_str = content.strip()

    try:
        data = json.loads(json_str)
        return CriteriaOutput.model_validate(data)
    except (json.JSONDecodeError, Exception) as e:
        # Fallback: return a minimal valid response
        return CriteriaOutput(
            criteria=[
                CriterionOutput(
                    name="Analysis Error",
                    description=f"Failed to parse AI response: {str(e)}",
                    score_a=50,
                    score_b=50,
                    weight=100,
                    sources=[],
                )
            ],
            winner="tie",
            reasoning="Unable to complete analysis due to parsing error.",
        )
```

"""Orchestrator combining web search and AI analysis."""

from datetime import datetime

from ..models import ComparisonResult, Criterion
from .tavily_service import search_both_options
from .claude_service import generate_criteria


async def generate_comparison(
    comparison_id: str,
    option_a: str,
    option_b: str,
    intent: str,
) -> ComparisonResult:
    """Orchestrate full comparison: web search -> AI analysis -> structured result.

    Flow:
    1. Search web for both options in parallel (Tavily)
    2. Generate criteria from research (Claude)
    3. Package into ComparisonResult
    """

    # Step 1: Parallel web search
    research_a, research_b = await search_both_options(option_a, option_b)

    # Check for complete search failure
    if research_a.error and research_b.error:
        return ComparisonResult(
            comparison_id=comparison_id,
            status="error",
            option_a_name=option_a,
            option_b_name=option_b,
            intent=intent,
            criteria=[],
            winner=None,
            reasoning=f"Web search failed: {research_a.error}",
            created_at=datetime.now(),
            completed_at=datetime.now(),
        )

    # Step 2: Generate criteria via Claude
    try:
        criteria_output = await generate_criteria(
            option_a=option_a,
            option_b=option_b,
            intent=intent,
            research_a=research_a,
            research_b=research_b,
        )
    except Exception as e:
        return ComparisonResult(
            comparison_id=comparison_id,
            status="error",
            option_a_name=option_a,
            option_b_name=option_b,
            intent=intent,
            criteria=[],
            winner=None,
            reasoning=f"AI analysis failed: {str(e)}",
            created_at=datetime.now(),
            completed_at=datetime.now(),
        )

    # Step 3: Convert to ComparisonResult format
    criteria = [
        Criterion(
            id=str(i + 1),
            name=c.name,
            description=c.description,
            score_a=c.score_a,
            score_b=c.score_b,
            weight=c.weight,
            sources=c.sources,
        )
        for i, c in enumerate(criteria_output.criteria)
    ]

    return ComparisonResult(
        comparison_id=comparison_id,
        status="completed",
        option_a_name=option_a,
        option_b_name=option_b,
        intent=intent,
        criteria=criteria,
        winner=criteria_output.winner,
        reasoning=criteria_output.reasoning,
        created_at=datetime.now(),
        completed_at=datetime.now(),
    )
```

Key patterns:
- Clean separation: orchestrator doesn't know Tavily/Claude internals
- Error handling at each step with meaningful messages
- Transforms AI output (CriteriaOutput) to app models (Criterion, ComparisonResult)
- Returns ComparisonResult directly (matches existing storage format)
