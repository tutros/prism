```typescript
import { useState, useMemo } from 'react'
import type { Criterion } from '../comparison/types'

export interface WeightedScores {
  totalA: number
  totalB: number
  winner: 'a' | 'b' | 'tie'
}

export function calculateScores(
  criteria: Criterion[],
  weights: Record<string, number>
): WeightedScores {
  if (criteria.length === 0) {
    return { totalA: 0, totalB: 0, winner: 'tie' }
  }

  // Normalize weights to sum to 100
  const totalWeight = Object.values(weights).reduce((sum, w) => sum + w, 0)
  const normalizer = totalWeight > 0 ? 100 / totalWeight : 1

  let totalA = 0
  let totalB = 0

  for (const criterion of criteria) {
    const weight = (weights[criterion.id] ?? criterion.weight) * normalizer / 100
    totalA += criterion.score_a * weight
    totalB += criterion.score_b * weight
  }

  // Round to 1 decimal
  totalA = Math.round(totalA * 10) / 10
  totalB = Math.round(totalB * 10) / 10

  let winner: 'a' | 'b' | 'tie'
  if (totalA > totalB) {
    winner = 'a'
  } else if (totalB > totalA) {
    winner = 'b'
  } else {
    winner = 'tie'
  }

  return { totalA, totalB, winner }
}

export function useWeightedScores(criteria: Criterion[]) {
  // Initialize weights from criteria defaults
  const [weights, setWeights] = useState<Record<string, number>>(() => {
    const initial: Record<string, number> = {}
    for (const c of criteria) {
      initial[c.id] = c.weight
    }
    return initial
  })

  const scores = useMemo(
    () => calculateScores(criteria, weights),
    [criteria, weights]
  )

  const updateWeight = (criterionId: string, value: number) => {
    setWeights((prev) => ({ ...prev, [criterionId]: value }))
  }

  return { weights, scores, updateWeight }
}
```

```tsx
interface WeightSliderProps {
  criterionId: string
  label: string
  value: number
  onChange: (criterionId: string, value: number) => void
}

export function WeightSlider({
  criterionId,
  label,
  value,
  onChange,
}: WeightSliderProps) {
  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    onChange(criterionId, parseInt(e.target.value, 10))
  }

  return (
    <div className="space-y-2">
      <div className="flex justify-between items-center">
        <label
          htmlFor={`weight-${criterionId}`}
          className="text-sm font-medium text-gray-700"
        >
          {label}
        </label>
        <span className="text-sm font-semibold text-blue-600 min-w-[3rem] text-right">
          {value}%
        </span>
      </div>
      <input
        id={`weight-${criterionId}`}
        type="range"
        min="0"
        max="100"
        step="1"
        value={value}
        onChange={handleChange}
        className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer accent-blue-600"
      />
    </div>
  )
}
```

```tsx
import { WeightSlider } from './WeightSlider'
import type { Criterion } from '../comparison/types'
import type { WeightedScores } from './useWeightedScores'

interface WeightsPanelProps {
  criteria: Criterion[]
  weights: Record<string, number>
  scores: WeightedScores
  optionAName: string
  optionBName: string
  onWeightChange: (criterionId: string, value: number) => void
}

export function WeightsPanel({
  criteria,
  weights,
  scores,
  optionAName,
  optionBName,
  onWeightChange,
}: WeightsPanelProps) {
  return (
    <div className="space-y-6">
      {/* Scores Summary */}
      <div className="p-4 bg-gray-50 rounded-lg">
        <h4 className="text-sm font-medium text-gray-500 mb-3">Weighted Scores</h4>
        <div className="grid grid-cols-2 gap-4">
          <div className={`p-3 rounded-lg ${scores.winner === 'a' ? 'bg-green-100 border-2 border-green-500' : 'bg-white border border-gray-200'}`}>
            <div className="text-sm text-gray-600">{optionAName}</div>
            <div className="text-2xl font-bold">{scores.totalA}</div>
          </div>
          <div className={`p-3 rounded-lg ${scores.winner === 'b' ? 'bg-green-100 border-2 border-green-500' : 'bg-white border border-gray-200'}`}>
            <div className="text-sm text-gray-600">{optionBName}</div>
            <div className="text-2xl font-bold">{scores.totalB}</div>
          </div>
        </div>
      </div>

      {/* Weight Sliders */}
      <div>
        <h4 className="text-sm font-medium text-gray-500 mb-3">Adjust Weights</h4>
        <div className="space-y-4">
          {criteria.map((criterion) => (
            <WeightSlider
              key={criterion.id}
              criterionId={criterion.id}
              label={criterion.name}
              value={weights[criterion.id] ?? criterion.weight}
              onChange={onWeightChange}
            />
          ))}
        </div>
      </div>
    </div>
  )
}
```

```tsx
import type { ComparisonResult } from '../comparison/types'

interface ResultsDisplayProps {

export function ResultsDisplay({ result }: ResultsDisplayProps) {

  if (result.status === 'processing') {
    return (
      <div className="flex items-center justify-center py-12">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600" />
        <span className="ml-3 text-gray-600">Analyzing options...</span>

      <div className="p-4 bg-red-50 border border-red-200 rounded-lg">

      </div>
    )

  // Use calculated winner instead of server's winner
  const winnerLabel = scores.winner === 'a'
    ? result.option_a_name
```
