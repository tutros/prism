```tsx
import { useState, useEffect } from 'react'
import { useComparison } from './useComparison'
import { ExamplePrompts, type Example } from './ExamplePrompts'

interface ComparisonFormProps {
  onSubmit: (comparisonId: string) => void
  initialValues?: Example | null
}

export function ComparisonForm({ onSubmit, initialValues }: ComparisonFormProps) {
  const [optionA, setOptionA] = useState('')
  const [optionB, setOptionB] = useState('')
  const [intent, setIntent] = useState('')
  const [showExamples, setShowExamples] = useState(true)

  const mutation = useComparison()

  // Apply initial values when provided
  useEffect(() => {
    if (initialValues) {
      setOptionA(initialValues.optionA)
      setOptionB(initialValues.optionB)
      setIntent(initialValues.intent)
      setShowExamples(false)
    }
  }, [initialValues])

  const handleExampleSelect = (example: Example) => {
    setOptionA(example.optionA)
    setOptionB(example.optionB)
    setIntent(example.intent)
    setShowExamples(false)
  }

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()

    if (!optionA.trim() || !optionB.trim() || !intent.trim()) {
      return
    }

    mutation.mutate(
      { option_a: optionA, option_b: optionB, intent },
      {
        onSuccess: (data) => {
          onSubmit(data.comparison_id)
        },
      }
    )
  }

  const handleClear = () => {
    setOptionA('')
    setOptionB('')
    setIntent('')
    setShowExamples(true)
  }

  return (
    <form onSubmit={handleSubmit} className="space-y-6">
      {showExamples && <ExamplePrompts onSelect={handleExampleSelect} />}

      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
        <div>
          <label htmlFor="optionA" className="block text-sm font-medium text-gray-700 mb-1">
            Option A
          </label>
          <input
            id="optionA"
            type="text"
            value={optionA}
            onChange={(e) => setOptionA(e.target.value)}
            placeholder="e.g., Glass bottle"
            className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
            required
          />
        </div>
        <div>
          <label htmlFor="optionB" className="block text-sm font-medium text-gray-700 mb-1">
            Option B
          </label>
          <input
            id="optionB"
            type="text"
            value={optionB}
            onChange={(e) => setOptionB(e.target.value)}
            placeholder="e.g., Plastic bottle"
            className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
            required
          />
        </div>
      </div>

      <div>
        <label htmlFor="intent" className="block text-sm font-medium text-gray-700 mb-1">
          Your Intent / Priorities
        </label>
        <textarea
          id="intent"
          value={intent}
          onChange={(e) => setIntent(e.target.value)}
          placeholder="e.g., I want to minimize environmental impact in my local community"
          rows={3}
          className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
          required
        />
      </div>

      <div className="flex flex-col sm:flex-row gap-3">
        <button
          type="submit"
          disabled={mutation.isPending}
          className="w-full sm:w-auto px-6 py-2 bg-blue-600 text-white font-medium rounded-lg hover:bg-blue-700 disabled:bg-blue-400 disabled:cursor-not-allowed transition-colors"
        >
          {mutation.isPending ? 'Comparing...' : 'Compare'}
        </button>

        {(optionA || optionB || intent) && (
          <button
            type="button"
            onClick={handleClear}
            className="w-full sm:w-auto px-6 py-2 bg-gray-100 text-gray-700 font-medium rounded-lg hover:bg-gray-200 transition-colors"
          >
            Clear
          </button>
        )}
      </div>

      {mutation.isError && (
        <p className="text-red-600 text-sm">
          Error: {mutation.error.message}
        </p>
      )}
    </form>
  )
}
```

```python
@router.get("/results")
async def list_results():
    """List all saved comparisons with metadata"""
    comparison_ids = storage.list_results()
    results = []
    for cid in comparison_ids:
        result = storage.read_result(cid)
        if result:
            results.append({
                "comparison_id": result.comparison_id,
                "option_a_name": result.option_a_name,
                "option_b_name": result.option_b_name,
                "status": result.status,
                "created_at": result.created_at.isoformat() if result.created_at else None,
            })
    # Sort by created_at descending (most recent first)
    results.sort(key=lambda x: x.get("created_at") or "", reverse=True)
    return {"comparisons": results}
```

```typescript
import { useQuery } from '@tanstack/react-query'
import { apiGet } from '../../lib/api'

interface ComparisonSummary {
  comparison_id: string
  option_a_name: string
  option_b_name: string
  status: 'processing' | 'completed' | 'error'
  created_at: string | null
}

interface HistoryResponse {
  comparisons: ComparisonSummary[]
}

export function useHistory() {
  return useQuery({
    queryKey: ['history'],
    queryFn: () => apiGet<HistoryResponse>('/api/results'),
    staleTime: 30000, // 30 seconds
  })
}

export type { ComparisonSummary }
```

```tsx
import { useHistory, type ComparisonSummary } from './useHistory'

interface HistoryListProps {
  onSelect: (comparisonId: string) => void
  currentId: string | null
}

export function HistoryList({ onSelect, currentId }: HistoryListProps) {
  const { data, isLoading, error } = useHistory()

  if (isLoading) {
    return (
      <div className="text-sm text-gray-500">Loading history...</div>
    )
  }

  if (error) {
    return (
      <div className="text-sm text-red-600">Failed to load history</div>
    )
  }

  const comparisons = data?.comparisons ?? []

  if (comparisons.length === 0) {
    return (
      <div className="text-sm text-gray-500">No previous comparisons</div>
    )
  }

  const formatDate = (dateStr: string | null) => {
    if (!dateStr) return ''
    const date = new Date(dateStr)
    return date.toLocaleDateString() + ' ' + date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
  }

  return (
    <div className="space-y-2">
      {comparisons.map((comparison) => (
        <button
          key={comparison.comparison_id}
          onClick={() => onSelect(comparison.comparison_id)}
          className={`w-full text-left p-3 rounded-lg border transition-colors ${
            currentId === comparison.comparison_id
              ? 'border-blue-500 bg-blue-50'
              : 'border-gray-200 bg-white hover:bg-gray-50'
          }`}
        >
          <div className="flex justify-between items-start">
            <div className="min-w-0 flex-1">
              <p className="font-medium text-gray-900 truncate">
                {comparison.option_a_name} vs {comparison.option_b_name}
              </p>
              <p className="text-xs text-gray-500 mt-1">
                {formatDate(comparison.created_at)}
              </p>
            </div>
            <span className={`ml-2 px-2 py-0.5 text-xs rounded ${
              comparison.status === 'completed'
                ? 'bg-green-100 text-green-700'
                : comparison.status === 'processing'
                ? 'bg-yellow-100 text-yellow-700'
                : 'bg-red-100 text-red-700'
            }`}>
              {comparison.status}
            </span>
          </div>
        </button>
      ))}
    </div>
  )
}
```

```tsx
import { useState } from 'react'
import { useQueryClient } from '@tanstack/react-query'
import { ComparisonForm } from './features/comparison/ComparisonForm'
import { ResultsDisplay } from './features/results/ResultsDisplay'
import { HistoryList } from './features/history/HistoryList'
import { useResults } from './features/results/useResults'
import type { Example } from './features/comparison/ExamplePrompts'

function App() {
  const [comparisonId, setComparisonId] = useState<string | null>(null)
  const [formKey, setFormKey] = useState(0)
  const queryClient = useQueryClient()
  const { data: result } = useResults(comparisonId)

  const handleNewComparison = (id: string) => {
    setComparisonId(id)
    // Invalidate history to include new comparison
    queryClient.invalidateQueries({ queryKey: ['history'] })
  }

  const handleSelectHistory = (id: string) => {
    setComparisonId(id)
  }

  const handleNewComparisonClick = () => {
    setComparisonId(null)
    setFormKey((k) => k + 1) // Force form reset
  }

  return (
    <div className="min-h-screen bg-gray-50">
      <div className="container mx-auto px-4 py-8 max-w-6xl">
        <header className="mb-8">
          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
            <div>
              <h1 className="text-2xl sm:text-3xl font-bold text-gray-900">
                Prism
              </h1>
              <p className="text-gray-600 text-sm sm:text-base">
                Compare options by surfacing hidden costs and systemic impacts.
              </p>
            </div>
            {comparisonId && (
              <button
                onClick={handleNewComparisonClick}
                className="self-start px-4 py-2 bg-blue-600 text-white text-sm font-medium rounded-lg hover:bg-blue-700 transition-colors"
              >
                New Comparison
              </button>
            )}
          </div>
        </header>

        <div className="grid grid-cols-1 lg:grid-cols-4 gap-8">
          {/* Sidebar - History */}
          <aside className="lg:col-span-1 order-2 lg:order-1">
            <div className="bg-white p-4 rounded-lg shadow">
              <h2 className="text-lg font-semibold mb-4">History</h2>
              <HistoryList
                onSelect={handleSelectHistory}
                currentId={comparisonId}
              />
            </div>
          </aside>

          {/* Main Content */}
          <main className="lg:col-span-3 space-y-8 order-1 lg:order-2">
            {!comparisonId && (
              <section className="bg-white p-4 sm:p-6 rounded-lg shadow">
                <h2 className="text-xl font-semibold mb-4">Compare Options</h2>
                <ComparisonForm
                  key={formKey}
                  onSubmit={handleNewComparison}
                />
              </section>
            )}

            {comparisonId && result && (
              <section className="bg-white p-4 sm:p-6 rounded-lg shadow">
                <h2 className="text-xl font-semibold mb-4">Results</h2>
                <ResultsDisplay result={result} />
              </section>
            )}
          </main>
        </div>
      </div>
    </div>
  )
}

export default App
```
