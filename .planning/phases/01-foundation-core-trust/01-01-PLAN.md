Create models.py with Pydantic v2 models:

```python
from pydantic import BaseModel, Field
from typing import Literal
from datetime import datetime

class ComparisonInput(BaseModel):
    option_a: str = Field(..., min_length=1, max_length=200)
    option_b: str = Field(..., min_length=1, max_length=200)
    intent: str = Field(..., min_length=1, max_length=1000)

class Criterion(BaseModel):
    id: str
    name: str
    description: str
    score_a: float = Field(ge=0, le=100)
    score_b: float = Field(ge=0, le=100)
    weight: float = Field(ge=0, le=100, default=50)
    sources: list[str] = []

class ComparisonResult(BaseModel):
    comparison_id: str
    status: Literal["processing", "completed", "error"]
    option_a_name: str
    option_b_name: str
    intent: str
    criteria: list[Criterion] = []
    winner: Literal["a", "b", "tie"] | None = None
    reasoning: str = ""
    created_at: datetime
    completed_at: datetime | None = None
```

Create storage.py with atomic JSON writes per research Pattern 3:

```python
from pathlib import Path
import json
import os
import tempfile
from .models import ComparisonResult

class JSONStorage:
    def __init__(self, data_dir: str = "data"):
        self.data_dir = Path(data_dir)
        self.data_dir.mkdir(exist_ok=True)

    def write_result(self, comparison_id: str, data: ComparisonResult) -> None:
        """Atomically write JSON to prevent corruption"""
        target_path = self.data_dir / f"{comparison_id}.json"

        # Write to temp file in same directory (required for atomic move)
        fd, tmp_path = tempfile.mkstemp(dir=self.data_dir, suffix='.tmp')
        try:
            with os.fdopen(fd, 'w') as tmp:
                json.dump(data.model_dump(mode='json'), tmp, indent=2, default=str)
                tmp.flush()
                os.fsync(tmp.fileno())
            # Atomic rename
            os.replace(tmp_path, target_path)
        except:
            if os.path.exists(tmp_path):
                os.unlink(tmp_path)
            raise

    def read_result(self, comparison_id: str) -> ComparisonResult | None:
        target_path = self.data_dir / f"{comparison_id}.json"
        if not target_path.exists():
            return None
        data = json.loads(target_path.read_text())
        return ComparisonResult.model_validate(data)

    def list_results(self) -> list[str]:
        """Return list of comparison IDs"""
        return [p.stem for p in self.data_dir.glob("*.json")]
```

Use `model_dump(mode='json')` for Pydantic v2 serialization (not `.dict()`).
Use `model_validate()` for deserialization (not `parse_obj()`).

import uuid
from datetime import datetime
import time

from ..models import ComparisonInput, ComparisonResult, Criterion
from ..storage import JSONStorage

router = APIRouter(prefix="/api", tags=["comparison"])
storage = JSONStorage()

    result = ComparisonResult(

@router.post("/compare")
async def submit_comparison(
    data: ComparisonInput,
    background_tasks: BackgroundTasks
):
    comparison_id = str(uuid.uuid4())

    # Write initial processing status
    initial = ComparisonResult(
        comparison_id=comparison_id,
        status="processing",
        option_a_name=data.option_a,
        option_b_name=data.option_b,
        intent=data.intent,
        criteria=[],
        winner=None,
        reasoning="",
        created_at=datetime.now(),
        completed_at=None
    )
    storage.write_result(comparison_id, initial)

    # Add mock processing to background
    background_tasks.add_task(generate_mock_result, comparison_id, data)

    return JSONResponse(
        status_code=202,
        content={"comparison_id": comparison_id}
    )

@router.get("/results/{comparison_id}")
async def get_results(comparison_id: str):
    result = storage.read_result(comparison_id)
    if not result:
        raise HTTPException(status_code=404, detail="Comparison not found")
    return result

@router.get("/results")
async def list_results():
    """List all saved comparison IDs"""
    return {"comparisons": storage.list_results()}

from fastapi.middleware.cors import CORSMiddleware

from .routers import health, comparison

    title="Hidden Cost Comparator API",
    description="Compare options by surfacing hidden costs and systemic impacts",
    version="0.1.0"
)

# CORS for Vite dev server
app.add_middleware(
    CORSMiddleware,

        "http://localhost:5173",
        "http://127.0.0.1:5173",
    ],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

app.include_router(health.router)
app.include_router(comparison.router)

@app.get("/")
async def root():
    return {"message": "Hidden Cost Comparator API", "docs": "/docs"}
