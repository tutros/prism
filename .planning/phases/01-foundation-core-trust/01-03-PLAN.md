const API_BASE = 'http://localhost:8000'

export async function apiPost<T>(endpoint: string, data: unknown): Promise<T> {
  const res = await fetch(`${API_BASE}${endpoint}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(data),
  })
  if (!res.ok) {
    throw new Error(`API error: ${res.status}`)
  }
  return res.json()
}

export async function apiGet<T>(endpoint: string): Promise<T> {
  const res = await fetch(`${API_BASE}${endpoint}`)
  if (!res.ok) {
    if (res.status === 404) {
      throw new Error('Not found')
    }
    throw new Error(`API error: ${res.status}`)
  }
  return res.json()
}

import { useState } from 'react'
import { useComparison } from './useComparison'

interface ComparisonFormProps {
  onSubmit: (comparisonId: string) => void
}

export function ComparisonForm({ onSubmit }: ComparisonFormProps) {
  const [optionA, setOptionA] = useState('')
  const [optionB, setOptionB] = useState('')
  const [intent, setIntent] = useState('')

  const mutation = useComparison()

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()

    if (!optionA.trim() || !optionB.trim() || !intent.trim()) {
      return
    }

    mutation.mutate(
      { option_a: optionA, option_b: optionB, intent },
      {
        onSuccess: (data) => {
          onSubmit(data.comparison_id)
        },
      }
    )
  }

  return (
    <form onSubmit={handleSubmit} className="space-y-6">
      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
        <div>
          <label htmlFor="optionA" className="block text-sm font-medium text-gray-700 mb-1">
            Option A
          </label>
          <input
            id="optionA"
            type="text"
            value={optionA}
            onChange={(e) => setOptionA(e.target.value)}
            placeholder="e.g., Glass bottle"
            className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
            required
          />
        </div>
        <div>
          <label htmlFor="optionB" className="block text-sm font-medium text-gray-700 mb-1">
            Option B
          </label>
          <input
            id="optionB"
            type="text"
            value={optionB}
            onChange={(e) => setOptionB(e.target.value)}
            placeholder="e.g., Plastic bottle"
            className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
            required
          />
        </div>
      </div>

      <div>
        <label htmlFor="intent" className="block text-sm font-medium text-gray-700 mb-1">
          Your Intent / Priorities
        </label>
        <textarea
          id="intent"
          value={intent}
          onChange={(e) => setIntent(e.target.value)}
          placeholder="e.g., I want to minimize environmental impact in my local community"
          rows={3}
          className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
          required
        />
      </div>

      <button
        type="submit"
        disabled={mutation.isPending}
        className="w-full md:w-auto px-6 py-2 bg-blue-600 text-white font-medium rounded-lg hover:bg-blue-700 disabled:bg-blue-400 disabled:cursor-not-allowed transition-colors"
      >
        {mutation.isPending ? 'Comparing...' : 'Compare'}
      </button>

      {mutation.isError && (
        <p className="text-red-600 text-sm">
          Error: {mutation.error.message}
        </p>
      )}
    </form>
  )
}

import { useQuery } from '@tanstack/react-query'
import { apiGet } from '../../lib/api'
import type { ComparisonResult } from '../comparison/types'

export function useResults(comparisonId: string | null) {
  return useQuery({
    queryKey: ['comparison', comparisonId],
    queryFn: async () => {
      if (!comparisonId) throw new Error('No comparison ID')
      return apiGet<ComparisonResult>(`/api/results/${comparisonId}`)
    },
    enabled: !!comparisonId,
    refetchInterval: (query) => {
      // Stop polling when completed or errored
      const data = query.state.data
      if (!data || data.status === 'completed' || data.status === 'error') {
        return false
      }
      return 1000 // Poll every 1 second
    },
  })
}

import type { ComparisonResult } from '../comparison/types'

interface ResultsDisplayProps {
  result: ComparisonResult
}

export function ResultsDisplay({ result }: ResultsDisplayProps) {
  if (result.status === 'processing') {
    return (
      <div className="flex items-center justify-center py-12">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600" />
        <span className="ml-3 text-gray-600">Analyzing options...</span>
      </div>
    )
  }

  if (result.status === 'error') {
    return (
      <div className="p-4 bg-red-50 border border-red-200 rounded-lg">
        <p className="text-red-700">An error occurred during analysis.</p>
      </div>
    )
  }

  const winnerLabel = result.winner === 'a'
    ? result.option_a_name
    : result.winner === 'b'
    ? result.option_b_name
    : 'Tie'

  return (
    <div className="space-y-6">
      {/* Winner Banner */}
      <div className="p-4 bg-green-50 border border-green-200 rounded-lg">
        <p className="text-lg font-semibold text-green-800">
          Winner: {winnerLabel}
        </p>
        <p className="text-sm text-green-700 mt-1">{result.reasoning}</p>
      </div>

      {/* Side-by-side Options */}
      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
        <div className={`p-4 rounded-lg border-2 ${result.winner === 'a' ? 'border-green-500 bg-green-50' : 'border-gray-200 bg-white'}`}>
          <h3 className="text-lg font-bold mb-2">{result.option_a_name}</h3>
          {result.winner === 'a' && (
            <span className="inline-block px-2 py-1 text-xs font-medium bg-green-500 text-white rounded mb-2">
              Winner
            </span>
          )}
        </div>
        <div className={`p-4 rounded-lg border-2 ${result.winner === 'b' ? 'border-green-500 bg-green-50' : 'border-gray-200 bg-white'}`}>
          <h3 className="text-lg font-bold mb-2">{result.option_b_name}</h3>
          {result.winner === 'b' && (
            <span className="inline-block px-2 py-1 text-xs font-medium bg-green-500 text-white rounded mb-2">
              Winner
            </span>
          )}
        </div>
      </div>

      {/* Criteria List */}
      <div className="space-y-4">
        <h3 className="text-lg font-semibold">Criteria</h3>
        {result.criteria.map((criterion) => (
          <div key={criterion.id} className="p-4 bg-white border border-gray-200 rounded-lg">
            <div className="flex justify-between items-start mb-2">
              <div>
                <h4 className="font-medium">{criterion.name}</h4>
                <p className="text-sm text-gray-600">{criterion.description}</p>
              </div>
            </div>
            <div className="grid grid-cols-2 gap-4 mt-3">
              <div>
                <span className="text-sm text-gray-500">{result.option_a_name}</span>
                <div className="text-lg font-semibold">{criterion.score_a}</div>
              </div>
              <div>
                <span className="text-sm text-gray-500">{result.option_b_name}</span>
                <div className="text-lg font-semibold">{criterion.score_b}</div>
              </div>
            </div>
            {criterion.sources.length > 0 && (
              <div className="mt-2 pt-2 border-t border-gray-100">
                <span className="text-xs text-gray-500">Sources: </span>
